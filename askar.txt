    // ============================================================
    // CONFIGURATION — Loaded dynamically from admin panel (KV)

    let RESULTS_RELEASED = false;
    let CONFIG = {};

    // ============================================================
    // Load config from API on page load
    // ============================================================
    async function init() {
      const form = document.getElementById('searchForm');
      const waiting = document.getElementById('waitingState');
      const btn = document.getElementById('btnSearch');

      try {
        const resp = await fetch('/api/config');
        CONFIG = await resp.json();
        RESULTS_RELEASED = CONFIG.resultsReleased || false;

        // Show announcement banner if set
        if (CONFIG.announcement) {
          const banner = document.createElement('div');
          banner.className = 'status-banner success show';
          banner.style.cssText = 'max-width:700px;margin:16px auto;display:flex;align-items:center;gap:10px;padding:14px 20px;';
          banner.innerHTML = '<span style="font-size:20px">📢</span><span>' + CONFIG.announcement + '</span>';
          const section = document.querySelector('.search-section');
          if (section) section.prepend(banner);
        }

        // Show ticker if set
        if (CONFIG.showTicker && CONFIG.tickerText) {
          const ticker = document.getElementById('newsTicker');
          const tickerContent = document.getElementById('tickerContent');
          tickerContent.innerHTML = '<span style="font-size:18px;animation:pulse 1s infinite">🚨</span> ' + CONFIG.tickerText;
          ticker.style.display = 'block';
        }

        // Telegram Subscription UI
        if (CONFIG.telegramBotUsername) {
          const tgArea = document.getElementById('telegramSubArea');
          const tgBtn = document.getElementById('telegramSubBtn');
          if (tgArea) tgArea.style.display = 'block';
          
          const subInput = document.getElementById('subNumber');
          const branchInput = document.getElementById('branch');
          const provinceInput = document.getElementById('province');
          
          if (subInput && tgBtn) {
            const updateTgLink = () => {
              const num = subInput.value.trim();
              const branch = branchInput ? branchInput.value : '';
              const province = provinceInput ? provinceInput.value : '';
              
              if (num && branch && province) {
                // Encode as: thanawiya_branch_province_number
                tgBtn.href = `https://t.me/${CONFIG.telegramBotUsername}?start=thanawiya_${branch}_${province}_${num}`;
              } else if (num) {
                // Fallback
                tgBtn.href = `https://t.me/${CONFIG.telegramBotUsername}?start=${num}`;
              } else {
                tgBtn.href = `https://t.me/${CONFIG.telegramBotUsername}`;
              }
            };
            subInput.addEventListener('input', updateTgLink);
            if (branchInput) branchInput.addEventListener('change', updateTgLink);
            if (provinceInput) provinceInput.addEventListener('change', updateTgLink);
            updateTgLink();
          }
        }
      } catch (e) {
        console.warn('Could not load config, using defaults');
      }

      if (!RESULTS_RELEASED) {
        btn.disabled = true;
        btn.innerHTML = '⏳ النتائج لم تصدر بعد';
        waiting.classList.add('show');
      } else {
        waiting.classList.remove('show');
        btn.disabled = false;
      }

      // Load stats
      const updateStatsUI = (statsData) => {
        if (!statsData.success) return;
        document.getElementById('activeUsers').textContent = statsData.active;
        document.getElementById('totalVisits').textContent = statsData.total.toLocaleString();
        
        if (statsData.countries) {
          const getFlag = (code) => {
            if (code === 'Unknown') return '🌐';
            return code.toUpperCase().replace(/./g, char => String.fromCodePoint(char.charCodeAt(0) + 127397));
          };
          const topFlags = Object.entries(statsData.countries)
            .sort((a, b) => b[1] - a[1])
            .slice(0, 4) // Show top 4 countries
            .map(c => getFlag(c[0]))
            .join(' ');
          document.getElementById('visitorFlags').textContent = topFlags || '🌐';
        }
      };

      try {
        await fetch('/api/stats', { method: 'POST' }); // Register visit
        const statsResp = await fetch('/api/stats');
        updateStatsUI(await statsResp.json());
      } catch (e) {
        console.warn('Could not load stats');
      }

      setInterval(async () => {
        try {
          const statsResp = await fetch('/api/stats');
          updateStatsUI(await statsResp.json());
        } catch (e) {}
      }, 30000);
    }

    // ============================================================
    // Handle search
    // ============================================================
    async function handleSearch(e) {
      e.preventDefault();

      if (!RESULTS_RELEASED) {
        showStatus('warning', 'النتائج لم تصدر بعد. يُرجى المتابعة على الموقع الرسمي لوزارة التربية والتعليم.');
        return;
      }

      const subNumber = document.getElementById('subNumber').value.trim();
      const studentName = document.getElementById('studentName').value.trim();
      const branch = document.getElementById('branch').value;
      const province = document.getElementById('province').value;

      if (!subNumber && !studentName) {
        showStatus('error', 'يُرجى إدخال رقم الاكتتاب أو الاسم الثلاثي.');
        return;
      }

      if (!branch) {
        showStatus('error', 'يُرجى اختيار الفرع (علمي / أدبي).');
        return;
      }

      // Show loading
      const btn = document.getElementById('btnSearch');
      btn.disabled = true;
      btn.innerHTML = '<div class="spinner"></div> جارٍ البحث...';
      hideResults();
      hideStatus();

      try {
        // Proxy through Cloudflare Function to ministry API
        const params = new URLSearchParams();
        if (subNumber) params.set('sub', subNumber);
        if (studentName) params.set('name', studentName);
        if (branch) params.set('branch', branch);
        if (province) params.set('province', province);

        const response = await fetch(`/api/results?${params.toString()}`);
        const result = await response.json();

        if (!result.success) {
          showStatus('error', result.error || 'لم يتم العثور على نتيجة بهذه البيانات.');
          return;
        }

        // Display the actual results from the ministry
        displayResults(result.data);
        showStatus('success', 'تم جلب النتيجة بنجاح من قاعدة بيانات وزارة التربية والتعليم.');

      } catch (err) {
        console.error('Search error:', err);
        showStatus('error', 'حدث خطأ أثناء الاتصال بالخادم. يُرجى المحاولة لاحقاً أو زيارة الموقع الرسمي <a href="https://moed.gov.sy" target="_blank" style="color:inherit;text-decoration:underline">moed.gov.sy</a>');
      } finally {
        btn.disabled = false;
        btn.innerHTML = `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round"><circle cx="11" cy="11" r="8"/><line x1="21" y1="21" x2="16.65" y2="16.65"/></svg> عرض النتيجة`;
      }
    }

    function displayResults(data) {
      const section = document.getElementById('resultsSection');
      const studentInfo = document.getElementById('studentInfo');
      const gradesBody = document.getElementById('gradesBody');
      const totalScore = document.getElementById('totalScore');

      // Student info
      studentInfo.innerHTML = `
        <div class="info-item"><span class="label">الاسم الثلاثي</span><span class="value">${data.name}</span></div>
        <div class="info-item"><span class="label">رقم الاكتتاب</span><span class="value">${data.subNumber}</span></div>
        <div class="info-item"><span class="label">الفرع</span><span class="value">${data.branch}</span></div>
        <div class="info-item"><span class="label">المحافظة</span><span class="value">${data.province}</span></div>
        <div class="info-item"><span class="label">المدرسة</span><span class="value">${data.school}</span></div>
        <div class="info-item"><span class="label">التقدير</span><span class="value">${data.grade}</span></div>
      `;

      // Grades
      gradesBody.innerHTML = data.subjects.map(s => `
        <div class="grade-row">
          <span class="subject">${s.name}</span>
          <span class="grade" style="color: ${s.score >= s.max * 0.4 ? 'var(--primary)' : 'var(--accent)'}">${s.score}</span>

    }
